cmake_minimum_required(VERSION 3.16)
project(NintendoRomAnalysis LANGUAGES C CXX)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(nes-native-c analysis/nes_native.c)
add_executable(nes-native-cpp analysis/nes_native.cpp)
add_executable(nes-rom-build-c analysis/nes_rom_build.c)
add_executable(nes-rom-build-cpp analysis/nes_rom_build.cpp)
add_executable(nes-disasm-c analysis/nes_disasm_native.c)
add_executable(nes-disasm-cpp analysis/nes_disasm_native.cpp)
if(MSVC)
  target_compile_options(nes-native-c PRIVATE /W4)
  target_compile_options(nes-native-cpp PRIVATE /W4)
  target_compile_options(nes-rom-build-c PRIVATE /W4)
  target_compile_options(nes-rom-build-cpp PRIVATE /W4)
  target_compile_options(nes-disasm-c PRIVATE /W4)
  target_compile_options(nes-disasm-cpp PRIVATE /W4)
else()
  target_compile_options(nes-native-c PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(nes-native-cpp PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(nes-rom-build-c PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(nes-rom-build-cpp PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(nes-disasm-c PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(nes-disasm-cpp PRIVATE -Wall -Wextra -Wpedantic)
endif()

add_executable(rom-manifest-c ../tools/rom_manifest.c)
add_executable(rom-manifest-cpp ../tools/rom_manifest.cpp)
add_executable(nes-chr-c ../graphics/nes_chr_viewer.c)
add_executable(nes-chr-cpp ../graphics/nes_chr_viewer.cpp)
add_executable(nes-apu-c ../audio/nes_apu_log.c)
add_executable(nes-apu-cpp ../audio/nes_apu_log.cpp)
if(MSVC)
  foreach(t rom-manifest-c rom-manifest-cpp nes-chr-c nes-chr-cpp nes-apu-c nes-apu-cpp)
    target_compile_options(${t} PRIVATE /W4)
  endforeach()
else()
  foreach(t rom-manifest-c rom-manifest-cpp nes-chr-c nes-chr-cpp nes-apu-c nes-apu-cpp)
    target_compile_options(${t} PRIVATE -Wall -Wextra -Wpedantic)
  endforeach()
endif()
